cmake_minimum_required(VERSION 3.16)
project(tinyvit_runner CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT MNN_SOURCE OR NOT MNN_BUILD)
    message(FATAL_ERROR "Set MNN_SOURCE and MNN_BUILD to the patched MNN source/build directories")
endif()
option(TINYVIT_STATIC "Link a self-contained Linux executable against static MNN" OFF)
if(TINYVIT_STATIC)
    set(CMAKE_FIND_LIBRARY_SUFFIXES ".a")
endif()
find_library(MNN_LIB MNN PATHS "${MNN_BUILD}" NO_DEFAULT_PATH REQUIRED)
add_executable(tinyvit_run tinyvit_run.cpp image_input.cpp)
target_include_directories(tinyvit_run PRIVATE "${MNN_SOURCE}/include" "${CMAKE_CURRENT_LIST_DIR}/..")
target_include_directories(tinyvit_run SYSTEM PRIVATE "${MNN_SOURCE}/3rd_party/imageHelper")
find_package(Threads REQUIRED)
target_link_libraries(tinyvit_run PRIVATE "${MNN_LIB}" Threads::Threads ${CMAKE_DL_LIBS} m)
target_compile_options(tinyvit_run PRIVATE -Wall -Wextra -Werror)
# After deployment, find libMNN.so beside the executable.
if(TINYVIT_STATIC)
    target_link_options(tinyvit_run PRIVATE -static)
else()
    set_target_properties(tinyvit_run PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
    add_custom_command(TARGET tinyvit_run POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${MNN_LIB}" "$<TARGET_FILE_DIR:tinyvit_run>/libMNN.so")
endif()
