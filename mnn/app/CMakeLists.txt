cmake_minimum_required(VERSION 3.16)
project(tinyvit_runner CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT MNN_SOURCE OR NOT MNN_BUILD)
    message(FATAL_ERROR "Set MNN_SOURCE and MNN_BUILD to the patched MNN source/build directories")
endif()

# Independent CPU/INT8-oracle/hardware diagnosis, outside benchmark timing.
add_executable(fc1_numeric_check fc1_numeric_check.cpp image_input.cpp)
target_include_directories(fc1_numeric_check PRIVATE "${MNN_SOURCE}/include" "${CMAKE_CURRENT_LIST_DIR}/.." "${CMAKE_CURRENT_LIST_DIR}/../../sw/include")
target_include_directories(fc1_numeric_check SYSTEM PRIVATE "${MNN_SOURCE}/schema/current" "${MNN_SOURCE}/3rd_party/flatbuffers/include" "${MNN_SOURCE}/3rd_party/imageHelper")
option(TINYVIT_STATIC "Link a self-contained Linux executable against static MNN" OFF)
if(TINYVIT_STATIC)
    set(CMAKE_FIND_LIBRARY_SUFFIXES ".a")
endif()
find_library(MNN_LIB MNN PATHS "${MNN_BUILD}" NO_DEFAULT_PATH REQUIRED)
add_executable(tinyvit_run tinyvit_run.cpp image_input.cpp operator_profile.cpp linear_override.cpp quantized_linear.cpp native_linear.cpp native_projection.cpp fused_attention_kernel.cpp fused_attention_regions.cpp gelu_lookup.cpp gelu_lookup_kernel.cpp)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^arm" AND NOT CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
    enable_language(ASM)
    target_sources(tinyvit_run PRIVATE gelu_reciprocal_arm32.S attention_arm32.S)
endif()
option(TINYVIT_BUILD_KERNEL_TESTS "Build independent attention and GELU numerical oracles" OFF)
if(TINYVIT_BUILD_KERNEL_TESTS)
    enable_testing()
    add_executable(tinyvit_attention_test test_fused_attention.cpp fused_attention_kernel.cpp)
    add_executable(tinyvit_gelu_test test_gelu.cpp gelu_lookup_kernel.cpp)
    foreach(kernel_test tinyvit_attention_test tinyvit_gelu_test)
        target_compile_options(${kernel_test} PRIVATE -Wall -Wextra -Werror)
        add_test(NAME ${kernel_test} COMMAND ${kernel_test})
    endforeach()
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "^arm" AND NOT CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
        target_sources(tinyvit_attention_test PRIVATE attention_arm32.S)
        target_sources(tinyvit_gelu_test PRIVATE gelu_reciprocal_arm32.S)
    endif()
endif()
target_include_directories(tinyvit_run PRIVATE "${MNN_SOURCE}/include" "${CMAKE_CURRENT_LIST_DIR}/..")
target_include_directories(tinyvit_run SYSTEM PRIVATE "${MNN_SOURCE}/3rd_party/imageHelper")
target_include_directories(tinyvit_run SYSTEM PRIVATE "${MNN_SOURCE}/schema/current" "${MNN_SOURCE}/3rd_party/flatbuffers/include")
target_include_directories(tinyvit_run SYSTEM PRIVATE "${MNN_SOURCE}/source")
find_package(Threads REQUIRED)
target_link_libraries(tinyvit_run PRIVATE "${MNN_LIB}" Threads::Threads ${CMAKE_DL_LIBS} m)
target_link_libraries(fc1_numeric_check PRIVATE "${MNN_LIB}" Threads::Threads ${CMAKE_DL_LIBS} m)
target_compile_options(fc1_numeric_check PRIVATE -Wall -Wextra -Werror -fno-rtti)
target_compile_options(tinyvit_run PRIVATE -Wall -Wextra -Werror)
# After deployment, find libMNN.so beside the executable.
if(TINYVIT_STATIC)
    target_link_options(tinyvit_run PRIVATE -static)
    target_link_options(fc1_numeric_check PRIVATE -static)
else()
    set_target_properties(fc1_numeric_check PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
    set_target_properties(tinyvit_run PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
    add_custom_command(TARGET tinyvit_run POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${MNN_LIB}" "$<TARGET_FILE_DIR:tinyvit_run>/libMNN.so")
endif()
